const express = require("express"); const path = require("path"); const crypto = require("crypto"); const multer = require("multer"); const cloudinary = require("cloudinary").v2; const app = express(); // ===================================================== // CLOUDINARY // ===================================================== cloudinary.config({ cloud_name: process.env.CLOUDINARY_CLOUD_NAME, api_key: process.env.CLOUDINARY_API_KEY, api_secret: process.env.CLOUDINARY_API_SECRET }); // ===================================================== // PAYMENT SCREENSHOT UPLOAD // ===================================================== const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: 5 * 1024 * 1024 }, fileFilter: (req, file, cb) => { if ( file.mimetype === "image/jpeg" || file.mimetype === "image/png" || file.mimetype === "image/webp" ) { cb(null, true); } else { cb( new Error( "Only JPG, PNG and WEBP images are allowed." ) ); } } }); // ===================================================== // EXPRESS // ===================================================== app.use( express.json({ limit: "1mb" }) ); app.use( express.urlencoded({ extended: true }) ); app.use( express.static(__dirname) ); // ===================================================== // ENVIRONMENT VARIABLES // ===================================================== const SHOPIFY_SHOP = process.env.SHOPIFY_SHOP; const SHOPIFY_CLIENT_ID = process.env.SHOPIFY_CLIENT_ID; const SHOPIFY_CLIENT_SECRET = process.env.SHOPIFY_CLIENT_SECRET; // ===================================================== // TEMPORARY CART STORAGE // ===================================================== const carts = new Map(); // ===================================================== // HOME // ===================================================== app.get("/", (req, res) => { res.send( "TopUpMandu Custom Checkout Backend is running!" ); }); // ===================================================== // CHECKOUT PAGE // ===================================================== app.get("/checkout", (req, res) => { res.sendFile( path.join( __dirname, "checkout.html" ) ); }); // ===================================================== // SHOPIFY TOKEN // ===================================================== async function getShopifyToken() { if ( !SHOPIFY_SHOP || !SHOPIFY_CLIENT_ID || !SHOPIFY_CLIENT_SECRET ) { throw new Error( "Shopify environment variables are missing." ); } const response = await fetch( `https://${SHOPIFY_SHOP}.myshopify.com/admin/oauth/access_token`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ client_id: SHOPIFY_CLIENT_ID, client_secret: SHOPIFY_CLIENT_SECRET, grant_type: "client_credentials" }) } ); const data = await response.json(); if (!response.ok) { throw new Error( data.error_description || data.error || "Could not get Shopify access token" ); } return data.access_token; } // ===================================================== // TEST SHOPIFY // ===================================================== app.get( "/test-shopify", async (req, res) => { try { await getShopifyToken(); res.json({ success: true, message: "Successfully connected to Shopify!" }); } catch (error) { console.error(error); res.status(500).json({ success: false, error: error.message }); } } ); // ===================================================== // SAVE CART // ===================================================== app.post( "/cart", (req, res) => { try { const cart = req.body; if ( !cart || !Array.isArray(cart.items) ) { return res.status(400).json({ success: false, error: "Invalid cart data." }); } if ( cart.items.length === 0 ) { return res.status(400).json({ success: false, error: "Cart is empty." }); } const cartId = crypto.randomUUID(); carts.set( cartId, { cart: cart, createdAt: Date.now() } ); // Remove carts older than 30 minutes for ( const [ id, stored ] of carts.entries() ) { if ( Date.now() - stored.createdAt > 30 * 60 * 1000 ) { carts.delete(id); } } res.json({ success: true, cartId: cartId }); } catch (error) { console.error(error); res.status(500).json({ success: false, error: error.message }); } } ); // ===================================================== // GET CART // ===================================================== app.get( "/cart/:cartId", (req, res) => { try { const stored = carts.get( req.params.cartId ); if (!stored) { return res.status(404).json({ success: false, error: "Cart not found or expired." }); } if ( Date.now() - stored.createdAt > 30 * 60 * 1000 ) { carts.delete( req.params.cartId ); return res.status(404).json({ success: false, error: "Cart has expired." }); } res.json({ success: true, cart: stored.cart }); } catch (error) { console.error(error); res.status(500).json({ success: false, error: error.message }); } } ); // ===================================================== // UPLOAD PAYMENT SCREENSHOT // ===================================================== app.post( "/upload-payment-screenshot", upload.single("paymentScreenshot"), async (req, res) => { try { if (!req.file) { return res.status(400).json({ success: false, error: "Payment screenshot is required." }); } const uploadResult = await new Promise( (resolve, reject) => { const stream = cloudinary.uploader.upload_stream( { folder: "topupmandu/payment-screenshots", resource_type: "image" }, ( error, result ) => { if (error) { reject(error); } else { resolve(result); } } ); stream.end( req.file.buffer ); } ); res.json({ success: true, screenshotUrl: uploadResult.secure_url }); } catch (error) { console.error( "Screenshot upload error:", error ); res.status(500).json({ success: false, error: error.message }); } } ); // ===================================================== // CREATE SHOPIFY ORDER // ===================================================== app.post( "/create-order", async (req, res) => { try { console.log( "======================================" ); console.log( "TOPUPMANDU CREATE ORDER" ); console.log( "======================================" ); // ================================================= // READ REQUEST // ================================================= let { customer, items, playerId, region, paymentMethod, paymentScreenshot } = req.body; // ================================================= // PARSE CUSTOMER / ITEMS IF SENT AS STRINGS // ================================================= let parsedCustomer = customer; let parsedItems = items; try { if ( typeof customer === "string" ) { parsedCustomer = JSON.parse(customer); } if ( typeof items === "string" ) { parsedItems = JSON.parse(items); } } catch (error) { return res.status(400).json({ success: false, error: "Invalid customer or cart data." }); } // ================================================= // VALIDATE ITEMS // ================================================= if ( !parsedItems || !Array.isArray(parsedItems) || parsedItems.length === 0 ) { return res.status(400).json({ success: false, error: "Cart items are required." }); } // ================================================= // CUSTOMER VALUES // ================================================= const customerName = String( parsedCustomer?.name || "" ).trim(); const customerEmail = String( parsedCustomer?.email || "" ).trim(); const customerPhone = String( parsedCustomer?.phone || "" ).trim(); // ================================================= // SPLIT CUSTOMER NAME // ================================================= let firstName = customerName; let lastName = ""; if (customerName.includes(" ")) { const nameParts = customerName.split( /\s+/ ); firstName = nameParts.shift(); lastName = nameParts.join(" "); } // ================================================= // SHOPIFY TOKEN // ================================================= const accessToken = await getShopifyToken(); // ================================================= // SHOPIFY LINE ITEMS // ================================================= const lineItems = parsedItems.map( (item) => { const variantId = item.variant_id || item.variantId; if (!variantId) { throw new Error( "Cart item is missing variant_id." ); } const variantGid = String( variantId ).startsWith( "gid://" ) ? String( variantId ) : `gid://shopify/ProductVariant/${variantId}`; // ----------------------------------------- // PRODUCT-SPECIFIC PROPERTIES // ----------------------------------------- const properties = []; /* * Keep the properties that came from the * Shopify cart item. * * This means products can have their own * Player ID / Region / Package information. */ if ( item.properties && typeof item.properties === "object" ) { for ( const [ key, value ] of Object.entries( item.properties ) ) { if ( value !== null && value !== undefined && String(value).trim() !== "" ) { properties.push({ name: String(key), value: String(value) }); } } } /* * If the checkout sends Player ID or Region * separately, add them only when they aren't * already present in the product properties. */ const hasPlayerId = properties.some( property => property.name .toLowerCase() .replace(/\s+/g, "") === "playerid" ); const hasRegion = properties.some( property => property.name .toLowerCase() .replace(/\s+/g, "") === "region" ); if ( playerId && !hasPlayerId ) { properties.push({ name: "Player ID", value: String( playerId ) }); } if ( region && !hasRegion ) { properties.push({ name: "Region", value: String( region ) }); } // ----------------------------------------- // PAYMENT METHOD // ----------------------------------------- if (paymentMethod) { properties.push({ name: "Payment Method", value: String( paymentMethod ) }); } // ----------------------------------------- // CREATE LINE ITEM // ----------------------------------------- const lineItem = { variantId: variantGid, quantity: Number( item.quantity ) || 1 }; if ( properties.length > 0 ) { lineItem.properties = properties; } return lineItem; } ); // ================================================= // ORDER CUSTOM ATTRIBUTES // ================================================= const attributes = []; // Payment Method if (paymentMethod) { attributes.push({ key: "Payment Method", value: String( paymentMethod ) }); } // Payment Screenshot if (paymentScreenshot) { attributes.push({ key: "Payment Screenshot", value: String( paymentScreenshot ) }); } // ================================================= // SHOPIFY ORDER // ================================================= const mutation = ` mutation CreateOrder( $order: OrderCreateOrderInput! ) { orderCreate( order: $order ) { userErrors { field message } order { id name displayFinancialStatus } } } `; // ================================================= // BUILD ORDER // ================================================= const order = { lineItems: lineItems, /* * Clean note. * * Customer name, payment method, * screenshot and player information * are no longer duplicated here. */ note: "TopUpMandu Custom Checkout", tags: [ "TopUpMandu", "Custom Checkout" ] }; // ================================================= // CUSTOMER EMAIL // ================================================= if (customerEmail) { order.email = customerEmail; } // ================================================= // CUSTOMER PHONE // ================================================= if (customerPhone) { order.phone = customerPhone; } // ================================================= // CUSTOMER // ================================================= if ( customerName || customerEmail || customerPhone ) { order.customer = { toUpsert: { ...(customerEmail ? { email: customerEmail } : {}), ...(customerPhone ? { phone: customerPhone } : {}), ...(firstName ? { firstName: firstName } : {}), ...(lastName ? { lastName: lastName } : {}) } }; } // ================================================= // CUSTOM ATTRIBUTES // ================================================= if ( attributes.length > 0 ) { order.customAttributes = attributes; } // ================================================= // DEBUG LOG // ================================================= console.log( "Shopify order payload:" ); console.log( JSON.stringify( order, null, 2 ) ); // ================================================= // SEND TO SHOPIFY // ================================================= const response = await fetch( `https://${SHOPIFY_SHOP}.myshopify.com/admin/api/2026-07/graphql.json`, { method: "POST", headers: { "Content-Type": "application/json", "X-Shopify-Access-Token": accessToken }, body: JSON.stringify({ query: mutation, variables: { order: order } }) } ); const result = await response.json(); console.log( "Shopify response:" ); console.log( JSON.stringify( result, null, 2 ) ); // ================================================= // GRAPHQL ERRORS // ================================================= if ( result.errors ) { return res.status(500).json({ success: false, error: result.errors }); } // ================================================= // ORDER RESULT // ================================================= const orderResult = result.data?.orderCreate; if ( !orderResult ) { return res.status(500).json({ success: false, error: "Shopify did not return an order." }); } // ================================================= // SHOPIFY USER ERRORS // ================================================= if ( orderResult.userErrors && orderResult.userErrors.length > 0 ) { return res.status(400).json({ success: false, errors: orderResult.userErrors }); } // ================================================= // SUCCESS // ================================================= console.log( "======================================" ); console.log( "ORDER CREATED SUCCESSFULLY" ); console.log( "Order:", orderResult.order?.name ); console.log( "======================================" ); res.json({ success: true, message: "Order created successfully!", screenshotUrl: paymentScreenshot || null, order: orderResult.order }); } catch (error) { console.error( "======================================" ); console.error( "CREATE ORDER ERROR:" ); console.error( error ); console.error( "======================================" ); res.status(500).json({ success: false, error: error.message }); } } ); // ===================================================== // MULTER ERROR HANDLER // ===================================================== app.use( ( error, req, res, next ) => { if ( error instanceof multer.MulterError ) { return res.status(400).json({ success: false, error: "Screenshot upload failed: " + error.message }); } if ( error && ( error.message === "Only JPG, PNG and WEBP images are allowed." || error.message === "Only image files are allowed." ) ) { return res.status(400).json({ success: false, error: error.message }); } next(error); } ); // ===================================================== // START SERVER // ===================================================== const PORT = process.env.PORT || 3000; app.listen( PORT, () => { console.log( `TopUpMandu backend running on port ${PORT}` ); } );
